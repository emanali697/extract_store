rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /jobs/{jobId}/video.mp4 {
      allow create: if request.resource.size <= 5 * 1024 * 1024 * 1024
                    && request.resource.contentType.matches('video/.*');
      allow read, update, delete: if false;
    }
  }
}
